Create `.github/workflows/ci.yaml` implementing the full trigger matrix from D-02 and D-03:

```yaml
# Single workflow, staged jobs, conditional execution per trigger (D-01)
# Job structure:
#   1. lint-test-build (push/PR) — lint + vitest + tsc build + path-filtered quick eval
#   2. full-eval (merge to main) — full eval:dev (60 fixtures) + eval:holdout (40 fixtures)
#   3. deploy-staging (merge to main) — wrangler deploy to afia-stage1.workers.dev
#   4. nightly-eval (schedule) — full eval:dev + eval:holdout for drift detection
#   5. signoff (manual) — human reviews staging JSONL output
#   6. edge-eval (schedule + merge) — CV pipeline on 20 edge-case images

name: CI

on:
  push:
    branches: ["**"]
  pull_request:
  schedule:
    - cron: "0 6 * * *"  # daily 06:00 UTC drift detection (D-02 nightly)
  workflow_dispatch:
    inputs:
      run_signoff:
        description: "Run signoff gate (manual human review)"
        type: boolean
        default: false

# Trust boundary: CI runner → Cloudflare API (deploy job needs wrangler auth)
# Trust boundary: CI runner → LLM API keys in env
env:
  GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
  GEMINI_API_KEYS: ${{ secrets.GEMINI_API_KEYS }}
  GEMINI_API_KEY2: ${{ secrets.GEMINI_API_KEY2 }}
  GEMINI_API_KEY3: ${{ secrets.GEMINI_API_KEY3 }}
  GROK_API_KEY: ${{ secrets.GROK_API_KEY }}
  MODEL_ID: ${{ secrets.MODEL_ID }}
  SUPABASE_URL: ${{ secrets.SUPABASE_URL }}
  SUPABASE_SERVICE_ROLE_KEY: ${{ secrets.SUPABASE_SERVICE_ROLE_KEY }}

concurrency:
  group: ${{ github.workflow }}-${{ github.ref }}
  cancel-in-progress: true

jobs:
  # ──────────────────────────────────────────────
  # JOB: lint-test-build (runs on push/PR)
  # ──────────────────────────────────────────────
  lint-test-build:
    if: github.event_name == 'push' || github.event_name == 'pull_request'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
      
      - run: pnpm install --frozen-lockfile
      
      - name: Build (tsc type-check)
        run: pnpm build
      
      - name: Test (vitest)
        run: pnpm test
      
      # Path filtering (D-03): quick eval only on worker/src/ or evals/ changes
      - uses: dorny/paths-filter@v3
        id: filter
        with:
          filters: |
            src_or_eval:
              - 'worker/src/**'
              - 'evals/**'
              - 'worker/test/fixtures/**'
      
      - name: Quick eval (10-fixture probe)
        if: steps.filter.outputs.src_or_eval == 'true'
        run: pnpm eval:dev-quick
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
      
      - name: Edge-case CV eval
        if: steps.filter.outputs.src_or_eval == 'true'
        run: pnpm eval:edge

  # ──────────────────────────────────────────────
  # JOB: full-eval (runs on merge to main)
  # ──────────────────────────────────────────────
  full-eval:
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    needs: lint-test-build
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      
      - name: Full dev eval (60 fixtures)
        run: pnpm eval:dev
      
      - name: Full holdout eval (40 fixtures)
        run: pnpm eval:holdout
      
      - name: Upload eval JSONL
        uses: actions/upload-artifact@v4
        with:
          name: eval-results-${{ github.sha }}
          path: runs/

  # ──────────────────────────────────────────────
  # JOB: edge-eval (CV pipeline on edge cases)
  # ──────────────────────────────────────────────
  edge-eval:
    if: (github.event_name == 'push' && github.ref == 'refs/heads/main') || github.event_name == 'schedule'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      - run: pnpm eval:edge

  # ──────────────────────────────────────────────
  # JOB: deploy-staging (deploys to afia-stage1.workers.dev)
  # ──────────────────────────────────────────────
  deploy-staging:
    if: github.event_name == 'push' && github.ref == 'refs/heads/main'
    needs: full-eval
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      
      - name: Deploy staging Worker
        run: npx wrangler deploy
        working-directory: worker
        env:
          CLOUDFLARE_API_TOKEN: ${{ secrets.CLOUDFLARE_API_TOKEN }}

  # ──────────────────────────────────────────────
  # JOB: nightly-eval (drift detection via cron)
  # ──────────────────────────────────────────────
  nightly-eval:
    if: github.event_name == 'schedule'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      - run: pnpm eval:dev
      - run: pnpm eval:holdout
      - run: pnpm eval:edge
      - name: Upload nightly eval results
        uses: actions/upload-artifact@v4
        with:
          name: nightly-eval-${{ github.run_id }}
          path: runs/

  # ──────────────────────────────────────────────
  # JOB: signoff (manual trigger, human review)
  # ──────────────────────────────────────────────
  signoff:
    if: github.event_name == 'workflow_dispatch' && inputs.run_signoff
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: pnpm/action-setup@v4
        with:
          version: 9
      - uses: actions/setup-node@v4
        with:
          node-version: 22
          cache: pnpm
      - run: pnpm install --frozen-lockfile
      - name: Run signoff gate (requires 3 prior holdout runs)
        run: pnpm signoff
        env:
          GEMINI_API_KEY: ${{ secrets.GEMINI_API_KEY }}
```

Implementation notes:
- Use `dorny/paths-filter@v3` for path detection — it's well-maintained and supports YAML filter syntax
- Concurrency group cancels in-flight runs on same branch (saves CI minutes)
- Use `--frozen-lockfile` to ensure reproducible installs
- Edge-eval is a separate job from full-eval because it runs the CV pipeline (no API key needed) vs LLM eval
- The deploy job runs wrangler from `worker/` directory (working-directory: worker)
- Upload eval artifacts so signoff gate review can access JSONL outputs
- Do NOT add `--env` flag to wrangler deploy — per D-07, staging uses the single existing config

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-01-01 | Information Disclosure | CI env vars | mitigate | All secrets stored in GitHub Actions secrets (masked in logs). No plaintext env in YAML. |
| T-01-02 | Spoofing | CI trigger (pull_request) | accept | PRs from forks trigger eval which costs ~$0.50. Acceptable for OSS. Path filtering limits to src/eval changes. |
| T-01-03 | Denial of Service | Nightly eval cost | accept | Nightly eval costs ~$5/run. Cron runs once daily. Acceptable ops cost. |
| T-01-04 | Elevation of Privilege | wrangler deploy token | mitigate | CLOUDFLARE_API_TOKEN stored as GitHub secret, scoped to afia-stage1 worker. Branch protection on main prevents unauthorized deploys. |
